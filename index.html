import pypdf
def extract_custom_pages(source_pdf, target_pages, output_name):
  reader = pypdf.PdfReader(source_pdf)
  writer = pypdf.PdfWriter()
  for p in target_pages:

  writer.write(output_name)
  print(f"Extracted {len(target_pages)} pages.")
